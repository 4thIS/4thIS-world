- 남의 소유 파일은 고치지 않는다. wj 소유: `apps/server/**`, `Dockerfile`, `docker-compose.yml`, `.env.example`, `docs/deploy.md`.

이 PR 고유 제약:

- 응답 타입 (웹 `api` 객체와 일치):

  | 경로                                                                              | 권한                   | 성공 응답                                                                                     |
  | --------------------------------------------------------------------------------- | ---------------------- | --------------------------------------------------------------------------------------------- |
  | `GET API.world`                                                                   | 누구나                 | `WorldResponse` — 승인 멤버만, 가입 순, 시간표는 (요일, 시작) 순, `serverTime` = `Date.now()` |
  | `GET API.me` · `PATCH API.me`                                                     | 로그인 멤버(상태 무관) | `Me`                                                                                          |
  | `DELETE API.me`                                                                   | 로그인 멤버            | `{ ok: true }`                                                                                |
  | `GET API.mySchedules` · `PUT API.mySchedules`                                     | 로그인 멤버(상태 무관) | `SchedulesResponse` (`{ items }`, 정렬됨)                                                     |
  | `GET API.adminMembers`                                                            | 관리자                 | `AdminMembersResponse` — 대기 먼저, 그다음 가입 순                                            |
  | `POST API.adminApprove(id)` · `API.adminReject(id)` · `PATCH API.adminMember(id)` | 관리자                 | `AdminMember`                                                                                 |
  | `DELETE API.adminMember(id)`                                                      | 관리자                 | `{ ok: true }`                                                                                |
  | `GET API.adminBackup`                                                             | 관리자                 | SQLite 파일 (`content-disposition: attachment; filename="world-YYYYMMDD.db"`)                 |

- 오류: 검증 실패 400 `invalid`, 로그인 안 함 401 `unauthorized`, 멤버가 관리자 API 401/403(guest 401, member 403), 관리자가 `/me` 401, 없는 멤버 404 `not_found`, 아이디 중복 409 `username_taken`, 시간표 겹침 **409 `overlap`, `detail = { indices: [앞, 뒤], items: [앞 일정, 뒤 일정] }`** (인덱스는 요청 `items` 배열 기준), 비밀번호 변경 때 현재 비밀번호가 틀리면 403 `forbidden`.
- 승인 대기 멤버도 자기 `/me`·시간표를 편집한다. `GET /world`에는 나오지 않는다.
- `app.events.emit('world:changed')`: 승인된 멤버가 시간표·생김새를 바꾸거나 탈퇴할 때, 관리자가 승인·거절·수정·삭제할 때. (대기 멤버 편집은 월드와 상관없어 내지 않는다.)
- 비밀번호를 바꾸면(본인) 지금 세션만 남기고 다른 세션을 끊는다. 관리자가 초기화하면 그 멤버의 세션을 모두 끊는다.

import type { FastifyInstance, LightMyRequestResponse } from 'fastify';
import { API, SessionSchema } from '@4this/shared';
import { buildApp } from './app';
import type { Config } from './config';
import { openDb, type Db } from './db';

/** 테스트 전용 설정 */
export function testConfig(overrides: Partial<Config> = {}): Config {
  return {
    port: 0,
    dbPath: ':memory:',
    adminPassword: 'admin-pass',
    sessionSecret: 'test-secret-test-secret-test-secret',
    webDist: null,
    production: false,
    trustProxy: false,
    ...overrides,
  };
}

export interface TestApp {
  app: FastifyInstance;
  db: Db;
  config: Config;
  close: () => Promise<void>;
}

/** 메모리 DB로 앱을 만든다 */
export async function makeTestApp(overrides: Partial<Config> = {}): Promise<TestApp> {
  const config = testConfig(overrides);
  const db = openDb(':memory:');
  const app = buildApp({ config, db });
  await app.ready();
  return {
    app,
    db,
    config,
    close: async () => {
      await app.close();
      db.close();
    },
  };
}

/** 응답의 Set-Cookie에서 sid를 꺼내 다음 요청의 cookie 헤더로 쓴다 */
export function sidCookie(res: LightMyRequestResponse): string {
  const cookie = res.cookies.find((c) => c.name === 'sid');
  if (!cookie) throw new Error('sid 쿠키가 없습니다');
  return `sid=${cookie.value}`;
}

export interface TestMember {
  id: string;
  cookie: string;
}

/** 가입하고 { id, cookie }를 돌려준다. approve면 DB에서 바로 승인한다 */
export async function signupAs(
  t: TestApp,
  username: string,
  { password = 'secret1', approve = false } = {},
): Promise<TestMember> {
  const res = await t.app.inject({
    method: 'POST',
    url: API.signup,
    payload: { username, password },
  });
  const session = SessionSchema.parse(res.json());
  if (session.kind !== 'member') throw new Error('가입 실패');
  if (approve)
    t.db.prepare("UPDATE members SET status = 'approved' WHERE id = ?").run(session.me.id);
  return { id: session.me.id, cookie: sidCookie(res) };
}

/** 관리자 로그인 쿠키 */
export async function adminCookie(t: TestApp): Promise<string> {
  const res = await t.app.inject({
    method: 'POST',
    url: API.adminLogin,
    payload: { password: t.config.adminPassword },
  });
  return sidCookie(res);
}

/** world:changed가 몇 번 났는지 센다 */
export function countWorldChanged(t: TestApp): () => number {
  let count = 0;
  t.app.events.on('world:changed', () => count++);
  return () => count;
}

import { LookSchema, type AdminMember, type Look, type Me, type MemberStatus } from '@4this/shared';
import type { Db } from './db';

/** members 테이블 한 줄 */
export interface MemberRow {
  id: string;
  username: string;
  password_hash: string;
  status: MemberStatus;
  look: string;
  created_at: number;
}

export function getMember(db: Db, id: string): MemberRow | undefined {
  return db.prepare('SELECT * FROM members WHERE id = ?').get(id) as MemberRow | undefined;
}

export function getMemberByUsername(db: Db, username: string): MemberRow | undefined {
  return db.prepare('SELECT * FROM members WHERE username = ?').get(username) as
    MemberRow | undefined;
}

export function insertMember(
  db: Db,
  m: { id: string; username: string; passwordHash: string; look: Look; createdAt: number },
): MemberRow {
  db.prepare(
    `INSERT INTO members (id, username, password_hash, status, look, created_at)
     VALUES (?, ?, ?, 'pending', ?, ?)`,
  ).run(m.id, m.username, m.passwordHash, JSON.stringify(m.look), m.createdAt);
  return getMember(db, m.id)!;
}

/** 바꿀 필드만 넘긴다. 바뀐 줄을 돌려주고, 멤버가 없으면 undefined */
export function updateMember(
  db: Db,
  id: string,
  patch: { username?: string; passwordHash?: string; status?: MemberStatus; look?: Look },
): MemberRow | undefined {
  const sets: string[] = [];
  const values: (string | number)[] = [];
  if (patch.username !== undefined) {
    sets.push('username = ?');
    values.push(patch.username);
  }
  if (patch.passwordHash !== undefined) {
    sets.push('password_hash = ?');
    values.push(patch.passwordHash);
  }
  if (patch.status !== undefined) {
    sets.push('status = ?');
    values.push(patch.status);
  }
  if (patch.look !== undefined) {
    sets.push('look = ?');
    values.push(JSON.stringify(patch.look));
  }
  if (sets.length > 0) {
    db.prepare(`UPDATE members SET ${sets.join(', ')} WHERE id = ?`).run(...values, id);
  }
  return getMember(db, id);
}

/** 멤버를 지운다. 시간표·세션은 ON DELETE CASCADE로 같이 지워진다 */
export function deleteMember(db: Db, id: string): boolean {
  return db.prepare('DELETE FROM members WHERE id = ?').run(id).changes > 0;
}

/** 승인된 멤버 (가입 순) */
export function listApprovedMembers(db: Db): MemberRow[] {
  return db
    .prepare("SELECT * FROM members WHERE status = 'approved' ORDER BY created_at, id")
    .all() as MemberRow[];
}

export function parseLook(row: MemberRow): Look {
  return LookSchema.parse(JSON.parse(row.look));
}

export function toMe(row: MemberRow): Me {
  return { id: row.id, username: row.username, status: row.status, look: parseLook(row) };
}

function scheduleCount(db: Db, id: string): number {
  return db.prepare('SELECT count(*) FROM schedules WHERE member_id = ?').pluck().get(id) as number;
}

export function toAdminMember(db: Db, row: MemberRow): AdminMember {
  return { ...toMe(row), createdAt: row.created_at, scheduleCount: scheduleCount(db, row.id) };
}

/** 관리자 화면 목록: 승인 대기 먼저, 그다음 가입 순 */
export function listAdminMembers(db: Db): AdminMember[] {
  const rows = db
    .prepare(
      `SELECT m.*, (SELECT count(*) FROM schedules s WHERE s.member_id = m.id) AS schedule_count
       FROM members m
       ORDER BY (m.status = 'pending') DESC, m.created_at, m.id`,
    )
    .all() as (MemberRow & { schedule_count: number })[];
  return rows.map((row) => ({
    ...toMe(row),
    createdAt: row.created_at,
    scheduleCount: row.schedule_count,
  }));
}

/** SQLite UNIQUE 제약 위반인지 (아이디 중복 경쟁 상태 대비) */
export function isUniqueViolation(error: unknown): boolean {
  return (
    error instanceof Error &&
    'code' in error &&
    (error as { code: unknown }).code === 'SQLITE_CONSTRAINT_UNIQUE'
  );
}

import type { Kind, ScheduleItem } from '@4this/shared';
import type { Db } from './db';

interface ScheduleRow {
  id: string;
  member_id: string;
  day: number;
  start: string;
  end: string;
  kind: Kind;
  title: string | null;

function toItem(row: ScheduleRow): ScheduleItem {
  const item: ScheduleItem = { day: row.day, start: row.start, end: row.end, kind: row.kind };
  if (row.title) item.title = row.title;
  return item;
}

/** 한 멤버의 시간표 (요일, 시작 순) */
export function listSchedules(db: Db, memberId: string): ScheduleItem[] {
  const rows = db
    .prepare('SELECT * FROM schedules WHERE member_id = ? ORDER BY day, "start"')
    .all(memberId) as ScheduleRow[];
  return rows.map(toItem);
}

/** 여러 멤버의 시간표를 한 번에 읽는다 (GET /world) */
export function listSchedulesByMember(db: Db): Map<string, ScheduleItem[]> {
  const rows = db
    .prepare('SELECT * FROM schedules ORDER BY member_id, day, "start"')
    .all() as ScheduleRow[];
  const byMember = new Map<string, ScheduleItem[]>();
  for (const row of rows) {
    const list = byMember.get(row.member_id) ?? [];
    list.push(toItem(row));
    byMember.set(row.member_id, list);
  }
  return byMember;
}

/** 시간표를 통째로 바꾼다. 하나의 트랜잭션이라 중간에 실패하면 원래대로 남는다 */
export function replaceSchedules(db: Db, memberId: string, items: readonly ScheduleItem[]): void {
  const remove = db.prepare('DELETE FROM schedules WHERE member_id = ?');
  const insert = db.prepare(
    'INSERT INTO schedules (id, member_id, day, "start", "end", kind, title) VALUES (?, ?, ?, ?, ?, ?, ?)',
  );
  db.transaction(() => {
    remove.run(memberId);
    for (const item of items) {
      insert.run(nanoid(), memberId, item.day, item.start, item.end, item.kind, item.title || null);
    }
  })();
}

import { randomBytes } from 'node:crypto';
import type { FastifyReply, FastifyRequest } from 'fastify';
import type { Session } from '@4this/shared';
import type { Config } from '../config';
import type { Db } from '../db';
import { forbidden, unauthorized } from '../errors';
import { getMember, toMe, type MemberRow } from '../members';

export const SESSION_COOKIE = 'sid';
export const SESSION_TTL_MS = 30 * 24 * 60 * 60 * 1000;

/** 요청을 보낸 사람 */
export type Auth =
  | { kind: 'guest' }
  | { kind: 'member'; sessionId: string; member: MemberRow }
  | { kind: 'admin'; sessionId: string };

export const GUEST: Auth = { kind: 'guest' };

interface SessionRow {
  id: string;
  member_id: string | null;
  is_admin: number;
  expires_at: number;
}

/** 무작위 토큰을 만들어 sessions에 저장하고 돌려준다 */
export function createSession(
  db: Db,
  owner: { memberId: string } | { admin: true },
  now = Date.now(),
): string {
  const id = randomBytes(32).toString('base64url');
  const memberId = 'memberId' in owner ? owner.memberId : null;
  db.prepare('INSERT INTO sessions (id, member_id, is_admin, expires_at) VALUES (?, ?, ?, ?)').run(
    id,
    memberId,
    memberId === null ? 1 : 0,
    now + SESSION_TTL_MS,
  );
  return id;
}

export function deleteSession(db: Db, id: string): void {
  db.prepare('DELETE FROM sessions WHERE id = ?').run(id);
}

/** 한 멤버의 세션을 모두 지운다 (비밀번호 변경·초기화). exceptId는 남긴다 */
export function deleteMemberSessions(db: Db, memberId: string, exceptId?: string): void {
  db.prepare('DELETE FROM sessions WHERE member_id = ? AND id != ?').run(memberId, exceptId ?? '');
}

/** 세션 토큰 → 사용자. 만료됐거나 없으면 guest (만료된 줄은 지운다) */
export function resolveSession(db: Db, id: string | null, now = Date.now()): Auth {
  if (!id) return GUEST;
  const row = db.prepare('SELECT * FROM sessions WHERE id = ?').get(id) as SessionRow | undefined;
  if (!row) return GUEST;
  if (row.expires_at <= now) {
    deleteSession(db, id);
    return GUEST;
  }
  if (row.is_admin) return { kind: 'admin', sessionId: row.id };
  const member = row.member_id ? getMember(db, row.member_id) : undefined;
  return member ? { kind: 'member', sessionId: row.id, member } : GUEST;
}

/** 서명된 sid 쿠키 값 → 세션 토큰. 위조됐으면 null */
export function readSessionToken(request: FastifyRequest): string | null {
  const raw = request.cookies[SESSION_COOKIE];
  if (!raw) return null;
  const unsigned = request.unsignCookie(raw);
  return unsigned.valid ? unsigned.value : null;
}

export function setSessionCookie(reply: FastifyReply, token: string, config: Config): void {
  reply.setCookie(SESSION_COOKIE, token, {
    path: '/',
    httpOnly: true,
    sameSite: 'lax',
    secure: config.production,
    signed: true,
    maxAge: SESSION_TTL_MS / 1000,
  });
}

export function clearSessionCookie(reply: FastifyReply): void {
  reply.clearCookie(SESSION_COOKIE, { path: '/' });
}

export function toSession(auth: Auth): Session {
  if (auth.kind === 'member') return { kind: 'member', me: toMe(auth.member) };
  if (auth.kind === 'admin') return { kind: 'admin' };
  return { kind: 'guest' };
}

/** 로그인한 멤버만 (상태 무관: 승인 대기·거절도 자기 정보는 본다) */
export function requireMember(request: FastifyRequest): MemberRow {
  if (request.auth.kind !== 'member') throw unauthorized();
  return request.auth.member;
}

/** 관리자만 */
export function requireAdmin(request: FastifyRequest): void {
  if (request.auth.kind === 'guest') throw unauthorized('관리자 로그인이 필요합니다');
  if (request.auth.kind !== 'admin') throw forbidden('관리자만 할 수 있습니다');
}

겹침은 shared `PutSchedulesRequestSchema`가 이미 거부하지만 그러면 400이 된다. 스펙은 409 + 겹친 일정 정보를 원하므로, 전체 검증 **전에** `ScheduleItemSchema.array()`로 항목만 읽어 `findOverlap`을 먼저 본다. 형식이 틀린 요청은 그다음 `parse`에서 400이 된다.

`apps/server/src/routes/me.ts`:

```ts
import type { FastifyInstance } from 'fastify';
import {
  API,
  findOverlap,
  PatchMeRequestSchema,
  PutSchedulesRequestSchema,
  ScheduleItemSchema,
  type Me,
  type SchedulesResponse,
} from '@4this/shared';
import { hashPassword, verifyPassword } from '../auth/password';
import { clearSessionCookie, deleteMemberSessions, requireMember } from '../auth/session';
import { forbidden, HttpError, parse } from '../errors';
import { deleteMember, toMe, updateMember, type MemberRow } from '../members';
import { listSchedules, replaceSchedules } from '../schedules';

/** 겹치는 시간표면 409 overlap. detail = { indices: [앞, 뒤], items: [앞 일정, 뒤 일정] } */
function assertNoOverlap(body: unknown): void {
  const items = (body as { items?: unknown } | null)?.items;
  const list = ScheduleItemSchema.array().safeParse(items);
  if (!list.success) return;
  const overlap = findOverlap(list.data);
  if (!overlap) return;
  const [a, b] = overlap;
  throw new HttpError(409, 'overlap', '겹치는 일정이 있습니다', {
    indices: [a, b],
    items: [list.data[a], list.data[b]],
  });
}

export function registerMeRoutes(app: FastifyInstance): void {
  const { db, events } = app;

  /** 승인된 멤버가 바뀌었을 때만 월드에 알린다 */
  const changed = (member: MemberRow) => {
    if (member.status === 'approved') events.emit('world:changed');
  };

  app.get(API.me, async (request): Promise<Me> => toMe(requireMember(request)));

  app.patch(API.me, async (request): Promise<Me> => {
    const member = requireMember(request);
    const body = parse(PatchMeRequestSchema, request.body);
    let passwordHash: string | undefined;
    if (body.password) {
      if (!(await verifyPassword(body.password.current, member.password_hash))) {
        throw forbidden('현재 비밀번호가 틀렸습니다');
      }
      passwordHash = await hashPassword(body.password.next);
    }
    const updated = updateMember(db, member.id, { look: body.look, passwordHash })!;
    if (passwordHash && request.auth.kind === 'member') {
      // 다른 기기의 세션은 끊고 지금 세션만 남긴다
      deleteMemberSessions(db, member.id, request.auth.sessionId);
    }
    if (body.look) changed(updated);
    return toMe(updated);
  });

  app.delete(API.me, async (request, reply) => {
    const member = requireMember(request);
    deleteMember(db, member.id);
    clearSessionCookie(reply);
    changed(member);
    return { ok: true as const };
  });

  app.get(API.mySchedules, async (request): Promise<SchedulesResponse> => {
    const member = requireMember(request);
    return { items: listSchedules(db, member.id) };
  });

  app.put(API.mySchedules, async (request): Promise<SchedulesResponse> => {
    const member = requireMember(request);
    assertNoOverlap(request.body);
    const { items } = parse(PutSchedulesRequestSchema, request.body);
    replaceSchedules(db, member.id, items);
    changed(member);
    return { items: listSchedules(db, member.id) };
  });
}

import fastifyCookie from '@fastify/cookie';
import fastifyRateLimit from '@fastify/rate-limit';
import Fastify, { type FastifyInstance } from 'fastify';
import { API } from '@4this/shared';
import { readSessionToken, resolveSession, type Auth } from './auth/session';
import type { Config } from './config';
import type { Db } from './db';
import { registerErrorHandlers } from './errors';
import { registerAdminRoutes } from './routes/admin';
import { registerAuthRoutes } from './routes/auth';
import { registerMeRoutes } from './routes/me';
import { registerWorldRoutes } from './routes/world';

/** 서버 안에서 주고받는 이벤트. wj#3의 소켓 계층이 world:changed를 받아 브로드캐스트한다 */
export type ServerEvents = { 'world:changed': [] };

declare module 'fastify' {
  interface FastifyInstance {
    config: Config;
    db: Db;
    events: EventEmitter<ServerEvents>;
  }
  interface FastifyRequest {
    auth: Auth;
  }
}

export interface AppOptions {
  config: Config;
  db: Db;
}

/** 서버 앱을 만든다. 테스트는 listen 없이 app.inject()로 부른다 */
export function buildApp({ config, db }: AppOptions): FastifyInstance {
  const app = Fastify({ logger: false, trustProxy: config.trustProxy });
  app.decorate('config', config);
  app.decorate('db', db);
  app.decorate('events', new EventEmitter<ServerEvents>());
  // 객체는 기본값으로 못 넣는다(Fastify 5). 자리만 잡고 onRequest 훅에서 채운다
  app.decorateRequest('auth');

  registerErrorHandlers(app);
  app.register(fastifyCookie, { secret: config.sessionSecret });
  app.register(fastifyRateLimit, { global: false });

  app.register(async (api) => {
    // 쿠키 sid → sessions → member/admin. 없거나 만료면 guest
    api.addHook('onRequest', async (request) => {
      request.auth = resolveSession(db, readSessionToken(request));
    });
    api.get(API.health, async () => ({ ok: true }));
    registerAuthRoutes(api);
    registerWorldRoutes(api);
    registerMeRoutes(api);
    registerAdminRoutes(api);
  });

  return app;
}

    const file = join(dir, 'world.db');
    try {
      await db.backup(file);
    } catch (error) {
      await rm(dir, { recursive: true, force: true });
      throw error;
    }
    const stream = createReadStream(file);
    stream.on('close', () => void rm(dir, { recursive: true, force: true }));
    return reply
      .header('content-type', 'application/vnd.sqlite3')
      .header('content-disposition', `attachment; filename="${backupFileName()}"`)
      .header('cache-control', 'no-store')
      .send(stream);
  });
}
```
